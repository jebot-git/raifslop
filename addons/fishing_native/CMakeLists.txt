cmake_minimum_required(VERSION 3.20)
project(fishing_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_DEFAULT_API_VERSION "4.7")
set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" CACHE FILEPATH "" FORCE)
add_subdirectory(${GODOT_CPP_PATH} godot-cpp)
add_library(fishing_native SHARED src/native.cpp src/codec.cpp)
if(FISHING_ZIG_GLIBC)
 target_sources(fishing_native PRIVATE src/zig_glibc_fini.cpp)
endif()
target_link_libraries(fishing_native PRIVATE godot-cpp)
target_compile_options(fishing_native PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3;-ffp-contract=off;-Wall;-Wextra>)
set_target_properties(fishing_native PROPERTIES PREFIX "lib")
